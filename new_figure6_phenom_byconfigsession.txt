SEARCH_DIRS <- c(DATA_DIR)


all_csv_files <- function() {

  files <- unlist(
    lapply(
      SEARCH_DIRS,
      function(d) {
        if (!dir.exists(d)) return(character(0))

        list.files(
          d,
          pattern = "\\.csv$",
          recursive = TRUE,
          full.names = TRUE
        )
      }
    )
  )

  unique(files[file.exists(files)])
}


normalise_header_name <- function(x) {
  x %>%
    stringr::str_trim() %>%
    stringr::str_to_lower() %>%
    stringr::str_replace_all("[^a-z0-9]+", "_") %>%
    stringr::str_replace_all("^_+|_+$", "")
}


# Recognises common variants:
#   vhq_1
#   vhq1
#   6d_vhq_1
#   vhq_1_1   (item 1, session 1)
#   vhq1_1
parse_vhq_name <- function(x) {

  z <- normalise_header_name(x)

  m <- stringr::str_match(
    z,
    "^(?:6d_?)?vhq_?([0-9]{1,2})(?:_([1-4]))?$"
  )

  tibble(
    original_name = x,
    normalised_name = z,
    item_number = suppressWarnings(as.integer(m[, 2])),
    session_suffix = suppressWarnings(as.integer(m[, 3]))
  ) %>%
    mutate(
      item_number = if_else(
        item_number %in% 1:18,
        item_number,
        NA_integer_
      )
    )
}


header_names <- function(path) {

  tryCatch(
    {
      h <- readr::read_csv(
        path,
        n_max = 0,
        show_col_types = FALSE,
        name_repair = "minimal"
      )

      names(h)
    },
    error = function(e) character(0)
  )
}


count_vhq_header <- function(path) {

  nms <- header_names(path)

  if (length(nms) == 0) return(0L)

  parsed <- parse_vhq_name(nms)

  sum(!is.na(parsed$item_number))
}


find_csv_by_substring <- function(
  substrings,
  required = TRUE,
  require_vhq = FALSE,
  label = "CSV"
) {

  all_files <- all_csv_files()

  hits <- character(0)

  # Preserve the original search-priority order, but collect all matches.
  for (substring in substrings) {

    matched <- all_files[
      stringr::str_detect(
        basename(all_files),
        stringr::regex(substring, ignore_case = TRUE)
      )
    ]

    hits <- c(hits, matched)
  }

  hits <- unique(hits)

  # If the filename search was too restrictive for a post-session file,
  # search all WP2/post-like CSVs and let VHQ-column detection choose.
  if (length(hits) == 0 && require_vhq) {
    hits <- all_files[
      stringr::str_detect(
        basename(all_files),
        stringr::regex("wp2.*post|post.*wp2", ignore_case = TRUE)
      )
    ]
  }

  if (length(hits) == 0) {
    if (required) {
      stop(
        "Could not find ",
        label,
        " matching any of:\n",
        paste(substrings, collapse = "\n")
      )
    }
    return(NULL)
  }

  info <- file.info(hits)

  candidates <- tibble(
    path = hits,
    file = basename(hits),
    filename_length = nchar(basename(hits)),
    modified = info$mtime,
    vhq_columns = if (require_vhq) {
      vapply(hits, count_vhq_header, integer(1))
    } else {
      0L
    }
  )

  if (require_vhq) {

    candidates <- candidates %>%
      arrange(
        desc(vhq_columns),
        filename_length,
        desc(modified)
      )

    cat(
      "\nCandidate ",
      label,
      " files, ranked by detected VHQ columns:\n",
      sep = ""
    )

    print(
      candidates %>%
        select(file, vhq_columns, modified),
      n = min(nrow(candidates), 20)
    )

    if (max(candidates$vhq_columns, na.rm = TRUE) == 0) {

      cat("\nNo strict VHQ columns were detected in filename-matched candidates.\n")
      cat("Columns containing 'vhq', 'visual', or 'halluc' in those files:\n\n")

      for (p in candidates$path) {

        nms <- header_names(p)

        maybe <- nms[
          stringr::str_detect(
            stringr::str_to_lower(nms),
            "vhq|visual|halluc"
          )
        ]

        cat("FILE: ", basename(p), "\n", sep = "")
        if (length(maybe) == 0) {
          cat("  [none]\n")
        } else {
          cat("  ", paste(maybe, collapse = "\n  "), "\n", sep = "")
        }
      }

      stop(
        "\nNo candidate post-session CSV contains detectable 6D-VHQ item columns. ",
        "The script has printed the candidate filenames and possible visual/VHQ fields above."
      )
    }

  } else {

    candidates <- candidates %>%
      arrange(
        filename_length,
        desc(modified)
      )
  }

  candidates$path[1]
}


read_qualtrics_real <- function(path) {

  x <- readr::read_csv(
    path,
    col_types = readr::cols(.default = readr::col_character()),
    show_col_types = FALSE
  )

  names(x) <- names(x) %>%
    stringr::str_trim() %>%
    stringr::str_to_lower()

  # Remove the two Qualtrics metadata/header rows where ResponseId is present.
  if ("responseid" %in% names(x)) {
    real_rows <- stringr::str_starts(
      as.character(x$responseid),
      "R_"
    )

    if (any(real_rows, na.rm = TRUE)) {
      x <- x[real_rows %in% TRUE, , drop = FALSE]
    }
  }

  # Harmonise participant ID.
  if (!"part_id" %in% names(x)) {

    alternatives <- c(
      "participantid",
      "participant_id",
      "externalreference",
      "external_reference",
      "id"
    )

    for (candidate in alternatives) {
      if (candidate %in% names(x)) {
        names(x)[names(x) == candidate] <- "part_id"
        break
      }
    }
  }

  if (!"part_id" %in% names(x)) {
    stop(
      "No participant ID column found in:\n",
      path
    )
  }

  x
}


clean_id <- function(x) {

  x %>%
    as.character() %>%
    stringr::str_trim() %>%
    stringr::str_to_lower() %>%
    dplyr::na_if("") %>%
    dplyr::na_if("nan") %>%
    dplyr::na_if("none") %>%
    dplyr::na_if('{"importid":"part_id"}') %>%
    stringr::str_extract("\\d{3,6}")
}


qnum <- function(x) {

  x <- stringr::str_trim(as.character(x))

  out <- suppressWarnings(as.numeric(x))

  needs_parse <- is.na(out) &
    !is.na(x) &
    x != "" &
    !tolower(x) %in% c("na", "nan", "none", "n/a")

  if (any(needs_parse)) {

    parsed <- stringr::str_match(
      x[needs_parse],
      "^\\s*([-+]?[0-9]+\\.?[0-9]*)"
    )[, 2]

    out[needs_parse] <- suppressWarnings(as.numeric(parsed))
  }

  out
}


find_first_col <- function(df, patterns) {

  nms <- names(df)

  for (pattern in patterns) {

    hit <- nms[
      stringr::str_detect(
        nms,
        stringr::regex(pattern, ignore_case = TRUE)
      )
    ]

    if (length(hit) > 0) {
      return(hit[1])
    }
  }

  NA_character_
}


# IMPORTANT:
# Do not use generic "SLS" as a condition matcher here.
# "Strobe-emphasis SLS" and "Wash-emphasis SLS" both contain "SLS".
normalise_registered_condition <- function(x) {

  z <- stringr::str_to_lower(
    stringr::str_trim(as.character(x))
  )

  dplyr::case_when(

    stringr::str_detect(
      z,
      "wash-emphasis|wash emphasis|control|ctrl|placebo|sham"
    ) ~ "Control",

    stringr::str_detect(
      z,
      "strobe-emphasis|strobe emphasis|intervention|treatment"
    ) ~ "Intervention",

    z %in% c("0", "c") ~ "Control",
    z %in% c("1", "i", "int") ~ "Intervention",

    TRUE ~ NA_character_
  )
}


registered_to_display <- function(x) {

  dplyr::case_when(
    x == "Intervention" ~ "Strobe-emphasis SLS",
    x == "Control"      ~ "Wash-emphasis SLS",
    TRUE                ~ NA_character_
  )
}


row_mean_available <- function(df_cols) {

  if (ncol(df_cols) == 0) {
    return(rep(NA_real_, nrow(df_cols)))
  }

  result <- rowMeans(df_cols, na.rm = TRUE)

  result[
    rowSums(!is.na(df_cols)) == 0
  ] <- NA_real_

  result
}

# ======================================================================
# 6. 6D-VHQ SCORING RULES
# ======================================================================

# 18 items total.
#
# Dimension composition:
#   Geometric Content = items 1–3
#   Semantic Content  = items 4–6
#   Detail            = items 7–9
#   Vividness         = items 10–12
#   Entropy           = items 13–15
#   Focality          = items 16–18
#
# Reverse-scored:
#   items 10, 11, 12, 17
#
# Original response scale = 1–5.
# Reverse transformation = 6 - response.

VHQ_DIMENSIONS <- list(
  "Geometric Content" = 1:3,
  "Semantic Content"  = 4:6,
  "Detail"            = 7:9,
  "Vividness"         = 10:12,
  "Entropy"           = 13:15,
  "Focality"          = 16:18
)

REVERSE_VHQ_ITEMS <- c(10, 11, 12, 17)


# ======================================================================
# 7. SCORE ONE SESSION
# ======================================================================

score_vhq_session <- function(
  df,
  session_number,
  force_session = FALSE
) {

  session_col <- find_first_col(
    df,
    c(
      "^session_n$",
      "^session$",
      "session_number",
      "sessionnum"
    )
  )

  sub <- df
  has_session_rows <- FALSE

  # Sessions 1–3 may be row-per-session.
  if (!is.na(session_col) && !force_session) {

    session_values <- qnum(df[[session_col]])

    has_session_rows <- any(
      session_values == session_number,
      na.rm = TRUE
    )

    if (has_session_rows) {

      sub <- df[
        !is.na(session_values) &
          session_values == session_number,
        ,
        drop = FALSE
      ]
    }
  }


  # Detect VHQ columns flexibly. The source exports used in the MRC project
  # can be row-per-session or wide/suffixed.
  vhq_map <- parse_vhq_name(names(sub)) %>%
    filter(!is.na(item_number))

  if (has_session_rows) {

    # Prefer unsuffixed item columns for row-per-session data.
    base_map <- vhq_map %>%
      filter(is.na(session_suffix))

    if (nrow(base_map) >= 1) {
      selected_map <- base_map
    } else {
      selected_map <- vhq_map %>%
        filter(session_suffix == session_number)
    }

  } else {

    # Prefer session-specific suffixed columns for wide data.
    suffix_map <- vhq_map %>%
      filter(session_suffix == session_number)

    if (nrow(suffix_map) >= 1) {
      selected_map <- suffix_map
    } else {
      selected_map <- vhq_map %>%
        filter(is.na(session_suffix))
    }
  }

  vhq_cols <- selected_map$original_name


  if (length(vhq_cols) == 0) {

    maybe_visual <- names(sub)[
      stringr::str_detect(
        stringr::str_to_lower(names(sub)),
        "vhq|visual|halluc"
      )
    ]

    stop(
      "No 6D-VHQ columns found for Session ",
      session_number,
      " in file: ",
      if ("source_file" %in% names(sub)) unique(sub$source_file)[1] else "<loaded post-session object>",
      "\nPossible visual/VHQ column names detected: ",
      if (length(maybe_visual) == 0) "[none]" else paste(maybe_visual, collapse = ", ")
    )
  }


  work <- sub %>%
    select(
      part_id,
      all_of(vhq_cols)
    )


  vhq_numeric <- work %>%
    select(all_of(vhq_cols)) %>%
    mutate(
      across(
        everything(),
        qnum
      )
    )


  # Recover the original VHQ item number from the already-parsed header map.
  selected_map <- selected_map[
    match(names(vhq_numeric), selected_map$original_name),
    ,
    drop = FALSE
  ]

  item_number <- selected_map$item_number


  # Reverse score 10, 11, 12, 17.
  for (j in seq_along(item_number)) {

    if (
      !is.na(item_number[j]) &&
      item_number[j] %in% REVERSE_VHQ_ITEMS
    ) {

      vhq_numeric[[j]] <- 6 - vhq_numeric[[j]]
    }
  }


  # Calculate three-item dimension means.
  scored <- tibble(
    part_id = work$part_id
  )

  for (dimension_name in names(VHQ_DIMENSIONS)) {

    wanted_items <- VHQ_DIMENSIONS[[dimension_name]]

    cols_for_dimension <- which(
      item_number %in% wanted_items
    )

    if (length(cols_for_dimension) > 0) {

      scored[[dimension_name]] <-
        row_mean_available(
          vhq_numeric[
            ,
            cols_for_dimension,
            drop = FALSE
          ]
        )

    } else {

      scored[[dimension_name]] <- NA_real_
    }
  }


  # Average duplicate participant rows within a session, if present.
  scored %>%
    mutate(
      session_n = session_number
    ) %>%
    group_by(
      part_id,
      session_n
    ) %>%
    summarise(
      across(
        all_of(names(VHQ_DIMENSIONS)),
        ~ {
          z <- suppressWarnings(
            as.numeric(.x)
          )

          if (all(is.na(z))) {
            NA_real_
          } else {
            mean(z, na.rm = TRUE)
          }
        }
      ),
      .groups = "drop"
    )
}

# ======================================================================
# 10. CONFIGURATION x SESSION DESCRIPTIVES + 95% CIs
# ======================================================================

vhq_summary <- vhq_long %>%
  group_by(
    dimension,
    configuration,
    session_n
  ) %>%
  summarise(

    n = n_distinct(part_id),

    mean = mean(
      score,
      na.rm = TRUE
    ),

    sd = sd(
      score,
      na.rm = TRUE
    ),

    se = sd / sqrt(n),

    ci95_low = if_else(
      n > 1,
      mean -
        qt(
          0.975,
          df = n - 1
        ) * se,
      NA_real_
    ),

    ci95_high = if_else(
      n > 1,
      mean +
        qt(
          0.975,
          df = n - 1
        ) * se,
      NA_real_
    ),

    .groups = "drop"
  ) %>%
  mutate(
    session_factor = factor(
      session_n,
      levels = 1:4
    ),
    configuration = factor(
      configuration,
      levels = CONFIG_ORDER
    ),
    dimension = factor(
      dimension,
      levels = VHQ_PLOT_ORDER
    )
  )


# Save the exact numbers plotted.
readr::write_csv(
  vhq_summary,
  CSV_FILE
)
